let audioCtx;

function ctxA(){ return audioCtx ||= new (window.AudioContext||window.webkitAudioContext)(); }

function gunSfx(vol=.18){

  const c=ctxA();

  const buf=c.createBuffer(1,Math.floor(c.sampleRate*.09),c.sampleRate), d=buf.getChannelData(0);

  for(let i=0;i<d.length;i++) d[i]=(Math.random()*2-1)*Math.pow(1-i/d.length,2.5);

  const s=c.createBufferSource(); s.buffer=buf;

  const g=c.createGain(); g.gain.value=vol;

  const f=c.createBiquadFilter(); f.type='lowpass'; f.frequency.value=2200;

  s.connect(f).connect(g).connect(c.destination); s.start();

}

function tone(freq,dur=.15,vol=.2,type='triangle'){

  const c=ctxA(), o=c.createOscillator(), g=c.createGain();

  o.type=type; o.frequency.setValueAtTime(freq,c.currentTime);

  o.frequency.exponentialRampToValueAtTime(Math.max(freq*.5,40),c.currentTime+dur);

  g.gain.setValueAtTime(vol,c.currentTime); g.gain.exponentialRampToValueAtTime(.001,c.currentTime+dur);

  o.connect(g).connect(c.destination); o.start(); o.stop(c.currentTime+dur);

}

    p.userData.life=.8; scene.add(p); particles.push(p);

  }

}

function inRect(x,z,r){ return x>r.x0&&x<r.x1&&z>r.z0&&z<r.z1; }

function groundHeight(x,z,feet){

  const inside=Math.abs(x)<12.8&&Math.abs(z)<12.8;

  const cands=[0];

  if(inside){ if(!inRect(x,z,HOLE1)) cands.push(F[1]); if(!inRect(x,z,HOLE2)) cands.push(F[2]); }

  if(x>9.0&&x<13&&z>-2.6&&z<10.6){ const t=Math.min(Math.max((10-z)/12,0),1); cands.push(F[0]+t*(F[1]-F[0])); }

  if(x>-13&&x<-9.0&&z>-10.6&&z<2.6){ const t=Math.min(Math.max((z+10)/12,0),1); cands.push(F[1]+t*(F[2]-F[1])); }

  platforms.forEach(p=>{ if(x>p.x0-.3&&x<p.x1+.3&&z>p.z0-.3&&z<p.z1+.3) cands.push(p.top); });

  let best=0; cands.forEach(h=>{ if(h<=feet+.6&&h>best) best=h; });

  return best;

}

function collide(){

  const feet=player.feetY, top=player.feetY+player.eye+.2;

  colliders.forEach(c=>{

    if(top<c.min.y||feet>c.max.y-.06) return;

    const nx=Math.max(c.min.x,Math.min(player.pos.x,c.max.x));

    const nz=Math.max(c.min.z,Math.min(player.pos.z,c.max.z));

    const dx=player.pos.x-nx, dz=player.pos.z-nz, dsq=dx*dx+dz*dz;

    if(dsq<player.r*player.r){ const dist=Math.sqrt(dsq)||.001, push=player.r-dist;

      player.pos.x+=dx/dist*push; player.pos.z+=dz/dist*push; }

  });

}

function startHost(){

  isHost=true;

  const code=makeCode();

  codeBox.textContent=code;

  statusEl.textContent='Verbinding voorbereiden...';

  peer=new Peer(PREFIX+code,{debug:0});

  peer.on('open',()=>statusEl.textContent='⏳ Wachten op je tegenstander...');

  peer.on('connection',c=>{ conn=c; wireConn(); });

  peer.on('error',err=>{

    if(err.type==='unavailable-id'){ peer.destroy(); startHost(); }  // code al bezet: nieuwe proberen

    else statusEl.textContent='⚠ Fout: '+err.type;

  });

}

function startJoin(code){

  if(code.length!==5){ status2El.textContent='⚠ Code moet 5 tekens zijn.'; return; }

  isHost=false;

  status2El.textContent='Verbinden...';

  peer=new Peer({debug:0});

  peer.on('open',()=>{

    conn=peer.connect(PREFIX+code,{reliable:true});

    wireConn();

    setTimeout(()=>{ if(!connected) status2El.textContent='⚠ Geen antwoord. Klopt de code? Is de host nog open?'; },6000);

  });

  peer.on('error',err=>status2El.textContent='⚠ Fout: '+err.type);

}

function wireConn(){

  conn.on('open',()=>{

    connected=true;

    remote.group.visible=true;

    readyTitle.textContent='✅ VERBONDEN';

    readyText.innerHTML='Je tegenstander is er. Eerste tot <b>'+WIN_SCORE+' kills</b> wint. Klik om te starten.';

    showStep(stepReady);

    setInterval(()=>{ if(connected) send({t:'ping',time:Date.now()}); },2000);

  });

  conn.on('data',onData);

  conn.on('close',()=>{ connected=false; onDisconnect(); });

  conn.on('error',()=>{ connected=false; onDisconnect(); });

}

function onDisconnect(){

  playing=false; document.exitPointerLock();

  remote.group.visible=false;

  readyTitle.textContent='🔌 VERBINDING VERBROKEN';

  readyText.textContent='Je tegenstander is weg. Herlaad de pagina om opnieuw te spelen.';

  document.getElementById('playBtn').disabled=true;

  showStep(stepReady); overlay.classList.remove('hidden');

}

function send(obj){ if(conn&&connected){ try{ conn.send(obj); }catch(e){} } }


function onData(m){

  switch(m.t){

    case 'st': // state van de tegenstander

      remote.pos.set(m.p[0],m.p[1],m.p[2]);

      remote.yaw=m.y; remote.sneak=m.s; remote.moving=m.m; remote.alive=m.a;

      remote.group.visible = m.a;

      break;

    case 'shot': { // hij schiet: flits + geluid op afstand

      remote.group.userData.flash.material.opacity=1;

      setTimeout(()=>remote.group.userData.flash.material.opacity=0,60);

      const d=remote.group.position.distanceTo(player.pos);

      gunSfx(Math.max(.02,.2-d*.006));

      break; }

    case 'hit': // hij claimt een treffer op mij

      if(!player.alive) break;

      player.hp-=m.dmg;

      damageFlash.style.opacity=1; setTimeout(()=>damageFlash.style.opacity=0,220);

      tone(110,.2,.25,'sawtooth');

      updateHUD();

      if(player.hp<=0) die(m.hs);

      break;

    case 'died': // hij is dood → punt voor mij

      myScore++;

      feed(m.hs?'💀 <b>HEADSHOT</b> — jij scoort!':'✔ Kill — jij scoort!','#7CFC00');

      showMsg(m.hs?'💀 HEADSHOT KILL':'✔ KILL','#7CFC00',900);

      tone(700,.2,.22);

      updateHUD();

      if(myScore>=WIN_SCORE) endMatch(true);

      break;

    case 'ping': send({t:'pong',time:m.time}); break;

    case 'pong': pingMs=Date.now()-m.time; break;

    case 'over': endMatch(m.win); break;

  }

}

const raycaster=new THREE.Raycaster();

function shoot(){

  if(ammo<=0){ if(!reloading){ reloading=1.7; tone(160,.1,.1,'square'); } return; }

  ammo--; recoil=1; gunSfx();

  weapon.userData.muzzle.material.opacity=1;

  setTimeout(()=>weapon.userData.muzzle.material.opacity=0,45);

  send({t:'shot'});


  const spread=(adsT>.6?.004:.022)*(sneaking?.6:1);

  raycaster.setFromCamera(new THREE.Vector2(rnd(-spread,spread),rnd(-spread,spread)),camera);

  const targets=[...solids];

  if(remote.group.visible && remote.alive) remote.group.traverse(c=>{ if(c.isMesh) targets.push(c); });

  const hits=raycaster.intersectObjects(targets);

  if(!hits.length) return;

  const h=hits[0];

  if(h.object.userData.isRemote){

    const hs = h.object.userData.part==='head';

    hitmarker.style.opacity=1; setTimeout(()=>hitmarker.style.opacity=0,90);

    burst(h.point,hs?0xff2222:0x883333,hs?14:6);

    tone(hs?520:300,.07,.12);

    send({t:'hit',dmg:hs?100:34,hs});

  } else burst(h.point,0x999999,4);

}


function die(headshot){

  player.alive=false; playing=true; // blijft in pointer lock, maar je kunt niets

  theirScore++;

  deadScreen.style.opacity=1;

  deadText.textContent = headshot ? '💀 HEADSHOT — je bent uitgeschakeld' : '☠ NEERGESCHOTEN';

  feed(headshot?'💀 Je kreeg een <b>headshot</b>':'☠ Je bent neergeschoten','#ff5c5c');

  tone(120,.5,.25,'sawtooth');

  respawnT=3;

  send({t:'died',hs:!!headshot});

  updateHUD();

  if(theirScore>=WIN_SCORE) setTimeout(()=>endMatch(false),400);

}

function respawn(){

  const s=SPAWNS[Math.floor(Math.random()*SPAWNS.length)];

  player.pos.set(s.x,0,s.z); player.feetY=F[s.f]; player.eye=EYE_STAND;

  player.hp=100; player.alive=true; ammo=30; reloading=0; velY=0;

  yaw=Math.atan2(-s.x,-s.z)+Math.PI; pitch=0;

  deadScreen.style.opacity=0;

  updateHUD();

}

function endMatch(win){

  document.getElementById('playBtn').disabled=true;

  showStep(stepReady); overlay.classList.remove('hidden');

  tone(win?600:120,.6,.25,win?'triangle':'sawtooth');

}

let netTimer=0;

const clock=new THREE.Clock();

function animate(){

  requestAnimationFrame(animate);

  const dt=Math.min(clock.getDelta(),.05), t=clock.elapsedTime;


  lampsList.forEach((L,i)=>{ if(L.flicker){

    const f=.7+Math.sin(t*17+i*3)*.15+(Math.random()<.02?-.5:0);

    L.light.intensity=L.base*Math.max(f,.15); } });


  if(playing && !matchOver){

    if(!player.alive){

      respawnT-=dt;

      deadText.textContent='☠ RESPAWN OVER '+Math.ceil(respawnT)+'s';

      if(respawnT<=0) respawn();

    } else {

      sneaking=!!keys['ShiftLeft']||!!keys['ShiftRight'];

      player.eye+=((sneaking?EYE_SNEAK:EYE_STAND)-player.eye)*Math.min(dt*9,1);

      adsT+=((adsing&&reloading<=0?1:0)-adsT)*Math.min(dt*11,1);

      camera.fov=75-adsT*30; camera.updateProjectionMatrix();

      crosshair.style.opacity=1-adsT*1.6;

      adsDot.style.opacity=adsT>.7?1:0;

      vignette.style.opacity=adsT*.8;


      fireCd-=dt;

      if(reloading>0){ reloading-=dt; if(reloading<=0){ reloading=0; ammo=30; tone(500,.1,.12); updateHUD(); } }

      else if(firing&&fireCd<=0){ fireCd=.092; shoot(); updateHUD(); }


      const speed=sneaking?3.6:(adsT>.5?5.2:8.5);

      const fwd=new THREE.Vector3(-Math.sin(yaw),0,-Math.cos(yaw));

      const right=new THREE.Vector3(Math.cos(yaw),0,-Math.sin(yaw));

      const move=new THREE.Vector3();

      if(keys['KeyW']) move.add(fwd);

      if(keys['KeyS']) move.sub(fwd);

      if(keys['KeyD']) move.add(right);

      if(keys['KeyA']) move.sub(right);

      if(move.lengthSq()) move.normalize().multiplyScalar(speed*dt);

      player.pos.add(move); collide();


      if(keys['Space']&&onGround){ velY=10.5; onGround=false; }

      velY-=26*dt; player.feetY+=velY*dt;

      const gh=groundHeight(player.pos.x,player.pos.z,player.feetY);

      if(player.feetY<=gh){ player.feetY=gh; velY=0; onGround=true; } else onGround=false;

      const dd=Math.hypot(player.pos.x,player.pos.z);

      if(dd>85){ player.pos.x*=85/dd; player.pos.z*=85/dd; }

      player.pos.y=player.feetY+player.eye;

    }


    // ~20x per seconde je positie doorsturen

    netTimer-=dt;

    if(netTimer<=0){

      netTimer=.05;

      const moving=!!(keys['KeyW']||keys['KeyA']||keys['KeyS']||keys['KeyD']);

      send({t:'st',p:[player.pos.x,player.feetY,player.pos.z],y:yaw,s:sneaking,m:moving,a:player.alive});

    }

    netinfo.textContent='verbonden · ping '+(pingMs||'—')+' ms';

  }


  // Tegenstander soepel interpoleren (anders teleporteert hij)

  if(remote.group.visible){

    remote.group.position.lerp(remote.pos, Math.min(dt*14,1));

    let dy=remote.yaw-remote.group.rotation.y;

    while(dy>Math.PI) dy-=Math.PI*2; while(dy<-Math.PI) dy+=Math.PI*2;

    remote.group.rotation.y += dy*Math.min(dt*14,1);

    const u=remote.group.userData;

    if(remote.moving){ u.walkPhase+=dt*(remote.sneak?5:11);

      u.legL.rotation.x=Math.sin(u.walkPhase)*.5; u.legR.rotation.x=-Math.sin(u.walkPhase)*.5; }

    else { u.legL.rotation.x*=.85; u.legR.rotation.x*=.85; }

    remote.group.scale.y = remote.sneak ? .72 : 1;   // hurkt hij, dan is hij kleiner

  }


  camera.position.copy(player.pos);

  camera.rotation.set(0,0,0);

  camera.rotateY(yaw); camera.rotateX(pitch);


  recoil=Math.max(0,recoil-dt*8);

  const moving=playing&&player.alive&&(keys['KeyW']||keys['KeyA']||keys['KeyS']||keys['KeyD']);

  const bob=moving?Math.sin(t*(sneaking?5:10))*(sneaking?.005:.012):Math.sin(t*2)*.004;

  const tgt=HIP.clone().lerp(ADSP,adsT);

  weapon.position.set(tgt.x,tgt.y+bob*(1-adsT*.7),tgt.z+recoil*.06);

  weapon.rotation.x=recoil*(adsT>.5?.05:.12);

  weapon.rotation.y=.03*(1-adsT);

  if(reloading>0){ weapon.rotation.x=.6; weapon.position.y=tgt.y-.12; }

  weapon.visible = player.alive;


  for(let i=particles.length-1;i>=0;i--){

    const p=particles[i];

    p.userData.vel.y-=14*dt;

    p.position.addScaledVector(p.userData.vel,dt);

    p.userData.life-=dt*1.4;

    p.scale.setScalar(Math.max(p.userData.life,.01));

    if(p.userData.life<=0){ scene.remove(p); particles.splice(i,1); }

  }

  renderer.render(scene,camera);

}

updateHUD();
